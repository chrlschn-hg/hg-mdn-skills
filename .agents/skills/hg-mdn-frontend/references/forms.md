# Forms and Validation

Forms use Nuxt UI's `UForm` + `UFormField` with a **Zod schema generated from the backend's OpenAPI document**.  The backend's limits (`maxLength`, `required`) reach the form automatically, so front-end validation cannot drift from what the server enforces.  The backend still validates and is the authority: the form catches mistakes before the round trip and shows the server's answer when the server disagrees.

## Where the schema comes from

Kubb's `pluginZod` writes a Zod schema per request body next to the types:

```ts
// src/api/generated/zod/createNoteRequestSchema.ts  (generated; do not edit)
export const createNoteRequestSchema = z.object({
  title: z.string().max(200).describe("Short title (required)."),
  body: z.string().max(20000).nullish().describe("Optional body text."),
  notebookId: z.uuid().nullish().describe("Optional notebook to file the note in."),
});
```

The limits come from the C# contract (`[MaxLength(NoteLimits.MaxTitleLength)]`), a nullable-with-default member becomes `.nullish()`, a `Guid` becomes `z.uuid()`.  Use the generated schema as the base and add only what the contract cannot say (for example "not blank", or a cross-field rule):

`src/app/src/components/NoteForm.vue`

```vue
<script setup lang="ts">
import type { FormSubmitEvent } from "@nuxt/ui";
import * as z from "zod";
import { createNoteRequestSchema } from "@/api/generated/zod/createNoteRequestSchema";
import type { NoteDraft } from "@/stores/notes-store";

// The limits (title <= 200, body <= 20000) come from the backend's OpenAPI document, generated by
// Kubb.  Add only what the contract cannot express: here, a blank title.
const schema = createNoteRequestSchema
  .pick({ title: true, body: true })
  .refine((note) => note.title.trim().length > 0, {
    path: ["title"],
    message: "Title is required.",
  });

type Schema = z.output<typeof schema>;

const props = defineProps<{
  saving?: boolean;
  /** The note being edited; omit to create a new one. */
  initial?: { title: string; body?: string };
}>();

const emit = defineEmits<{ submit: [draft: NoteDraft] }>();

// Plain strings: inputs bind to `string`, while the schema also allows null/undefined for `body`.
const state = reactive({ title: "", body: "" });
const form = useTemplateRef("form");

/** Fills the fields from `initial`: the note's values when editing, empty when creating. */
function fill(initial: typeof props.initial) {
  state.title = initial?.title ?? "";
  state.body = initial?.body ?? "";
}

// Create: the fields start empty.  Edit: they are refilled whenever a different note is passed in
// (the user opens another note, or the save came back and the parent passes the updated one).
watch(() => props.initial, fill, { immediate: true });

/** Which field shows each of the backend's error codes. */
const fieldForCode: Record<string, keyof Schema> = {
  invalid_title: "title",
  title_too_long: "title",
  body_too_long: "body",
};

/** Unsaved changes: the fields differ from what the form started with (empty, or `initial`). */
const isDirty = computed(
  () =>
    state.title.trim() !== (props.initial?.title ?? "").trim() ||
    state.body.trim() !== (props.initial?.body ?? "").trim(),
);

function onSubmit(event: FormSubmitEvent<Schema>) {
  emit("submit", { title: event.data.title.trim(), body: event.data.body?.trim() || undefined });
}

/**
 * Shows a failed save on the field the backend's error code names.  Returns `false` when the code
 * belongs to no field, so the caller can show it elsewhere (a toast).
 */
function showFailure(failure: { code: string | null; message: string }): boolean {
  const name = failure.code ? fieldForCode[failure.code] : undefined;

  if (!name) {
    return false;
  }

  form.value?.setErrors([{ name, message: failure.message }]);
  return true;
}

/** Restores the starting values (empty, or `initial`) and clears errors, e.g. after a save. */
function reset() {
  fill(props.initial);
  form.value?.clear();
}

defineExpose({ showFailure, reset, isDirty });
</script>

<template>
  <UForm
    ref="form"
    :schema="schema"
    :state="state"
    class="space-y-4"
    :disabled="saving"
    @submit="onSubmit"
  >
    <UFormField label="Title" name="title" required>
      <UInput v-model="state.title" placeholder="What to remember" class="w-full" />
    </UFormField>

    <UFormField label="Body" name="body" description="Optional.">
      <UTextarea v-model="state.body" :rows="4" class="w-full" />
    </UFormField>

    <UButton
      type="submit"
      :icon="initial ? 'i-hugeicons-checkmark-circle-02' : 'i-hugeicons-plus-sign'"
      :loading="saving"
    >
      {{ initial ? "Save changes" : "Add note" }}
    </UButton>
  </UForm>
</template>
```

How the form works:

- **`UForm`** takes `:schema` and `:state` and validates on blur, change, and input (submit always validates).  `@submit` fires only for **valid** data and receives `event.data` (the schema's output).
- **`UFormField name="title"`**: the `name` ties the field to its error; nested paths use dots (`address.city`).  A `label`, `description` (help text), and `required` marker belong on the field, not on the input.
- **State is plain strings** (`reactive({ title: "", body: "" })`) because inputs bind to `string`; the schema may allow `null`/`undefined`.  Normalize on submit (`trim()`, empty body -> `undefined`).
- **`:disabled="saving"`** on the form disables every input during a save.
- Validation **messages** are Zod's defaults (`Too big: expected string to have <=200 characters`); override them in the schema or with `z.config` when wording matters.  The Kubb `describe(...)` text is the C# XML doc: handy as field help text.

## Showing the server's answer

The backend returns an `ApiError` with a **stable code** for expected failures (`title_too_long`, `invalid_title`).  The store returns it in a `SaveResult` (`api-client.md`); the form maps the code to the field that owns it (`fieldForCode` and `showFailure` above).

- `setErrors([{ name, message }])` on the `UForm` template ref puts the message on that field exactly like a client-side error; `clear()` removes errors.
- **The form emits a draft; the page saves it.**  The form exposes `showFailure`, `reset`, and `isDirty` with `defineExpose`, and the **page** (which owns the store) calls them after the save returns.  `defineExpose` is the one sanctioned downward command, for results that arrive after an emit; anything declarative is a prop.
- On success the page calls `reset()`, closes the container, and toasts; on failure it calls `showFailure(result)` and toasts only when that returns `false` (the code matches no field).  Never silently drop a failure.  The notes page (`lists-tables.md`) shows `onSubmit`.

## Create and edit in one form

An optional `initial` prop makes the same form an edit form: with no `initial` it creates, with `initial` it edits that entity.  The local state refills from `initial` whenever it changes (the user opens another note, or the save came back and the parent passes the updated one), which is the `watch(() => props.initial, fill, { immediate: true })` above.

- **Dirty is relative to the starting values** (empty, or `initial`), so an untouched edit form does not ask to discard.
- `reset()` restores the starting values and clears errors; the page calls it after a successful save.
- Edit and create share validation, error mapping, and layout; only the button label and starting values differ.  The slideover passes `initial` through and titles itself "Edit note" or "New note".
- Tests: starts from the note's values and is clean; becomes dirty when changed; refills when `initial` changes; `reset` restores the note, not an empty form.

## Rules

1. **Schema from the contract** (`@/api/generated/zod/...`); never retype a limit in the front-end.  Extend with `.refine(...)`/`.extend(...)` for UI-only rules.
2. **Forms never use a store or the API client** (a `type` import of the draft is fine; `components-composables.md`).
3. **Every input sits in a `UFormField`** with a `name` and `label`.  A placeholder is not a label.
4. **Disable while saving** (`:disabled="saving"` on the form, `:loading="saving"` on the submit button).
5. **Reset after success**, keep the typed values after failure.
6. **Unsaved changes are protected** when the form sits in something closable: expose `isDirty` and confirm (`overlays.md`, "Unsaved changes").

A form with no contract (no request body) may use `:validate="(state) => errors"`; otherwise use Zod, because Kubb generates it.

## Testing a form

Mount it with `mountWithUi`, set values, submit, and flush promises (`UForm` validates asynchronously):

`src/app/src/components/NoteForm.test.ts`

```ts
import { flushPromises } from "@vue/test-utils";
import { describe, expect, it } from "vitest";
import { mountWithUi } from "@/test/mount";
import NoteForm from "./NoteForm.vue";

async function submit(wrapper: ReturnType<typeof mountWithUi<typeof NoteForm>>, title: string) {
  await wrapper.find("input").setValue(title);
  await wrapper.find("form").trigger("submit");
  await flushPromises();
}

describe("NoteForm", () => {
  it("emits the trimmed draft when valid", async () => {
    const wrapper = mountWithUi(NoteForm);

    await wrapper.find("input").setValue("  Hello  ");
    await wrapper.find("textarea").setValue("World");
    await wrapper.find("form").trigger("submit");
    await flushPromises();

    expect(wrapper.emitted("submit")).toEqual([[{ title: "Hello", body: "World" }]]);
  });

  it("blocks a blank title and says why", async () => {
    const wrapper = mountWithUi(NoteForm);

    await submit(wrapper, "   ");

    expect(wrapper.emitted("submit")).toBeUndefined();
    expect(wrapper.text()).toContain("Title is required.");
  });

  it("enforces the limit that comes from the backend contract", async () => {
    const wrapper = mountWithUi(NoteForm);

    await submit(wrapper, "a".repeat(201));

    expect(wrapper.emitted("submit")).toBeUndefined();
    expect(wrapper.text()).toMatch(/200/);
  });

  it("shows a backend error on the field its code names", async () => {
    const wrapper = mountWithUi(NoteForm);

    const shown = wrapper.vm.showFailure({
      code: "title_too_long",
      message: "Title may be at most 200 characters.",
    });
    await flushPromises();

    expect(shown).toBe(true);
    expect(wrapper.text()).toContain("Title may be at most 200 characters.");
  });

  it("leaves a code that belongs to no field for the caller to show", () => {
    const wrapper = mountWithUi(NoteForm);

    expect(wrapper.vm.showFailure({ code: "notebook_not_found", message: "Gone." })).toBe(false);
    expect(wrapper.vm.showFailure({ code: null, message: "Network error." })).toBe(false);
  });

  it("edits an existing note: starts from its values and is clean until changed", async () => {
    const wrapper = mountWithUi(NoteForm, {
      props: { initial: { title: "Groceries", body: "Milk" } },
    });

    expect((wrapper.find("input").element as HTMLInputElement).value).toBe("Groceries");
    expect((wrapper.find("textarea").element as HTMLTextAreaElement).value).toBe("Milk");
    expect(wrapper.vm.isDirty).toBe(false);
    expect(wrapper.text()).toContain("Save changes");

    await wrapper.find("input").setValue("Groceries and more");

    expect(wrapper.vm.isDirty).toBe(true);
  });

  it("refills when a different note is passed in", async () => {
    const wrapper = mountWithUi(NoteForm, { props: { initial: { title: "First" } } });

    await wrapper.setProps({ initial: { title: "Second", body: "Other" } });

    expect((wrapper.find("input").element as HTMLInputElement).value).toBe("Second");
    expect((wrapper.find("textarea").element as HTMLTextAreaElement).value).toBe("Other");
  });

  it("reset restores the note's values, not an empty form", async () => {
    const wrapper = mountWithUi(NoteForm, { props: { initial: { title: "Keep me" } } });
    await wrapper.find("input").setValue("changed");

    wrapper.vm.reset();
    await flushPromises();

    expect((wrapper.find("input").element as HTMLInputElement).value).toBe("Keep me");
    expect(wrapper.vm.isDirty).toBe(false);
  });

  it("reports unsaved text as dirty", async () => {
    const wrapper = mountWithUi(NoteForm);
    expect(wrapper.vm.isDirty).toBe(false);

    await wrapper.find("input").setValue("draft");

    expect(wrapper.vm.isDirty).toBe(true);
  });
});
```

Assert: valid input emits the normalized draft; each rule blocks the submit and shows its message; the contract's limit is enforced (assert `/200/`, not Zod's exact wording); `showFailure` returns `true`/`false` and shows the message; `isDirty` follows typing; with `initial` the form is filled and clean until changed, refills when `initial` changes, and `reset` restores it.
